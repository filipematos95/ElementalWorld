% LEVEL 4: PROCESSES & UPDATING SUBMODELS

\node[procbox, below left=1.2cm and -1.0cm of entities, text width=7.2cm] (p1_p4) {
    \textbf{Processes 1--4: Physical \& Physiological Setup}\\
    \hrulefill\\[2pt]
    \textbf{[P1] Diffusion \& Inflow:}\\
    $I^{\text{diff}} = \text{Conv2D}(\mathbf{I}, K), \quad I^{\text{new}} = I^{\text{diff}} + r^{\text{in}} I_{\text{rate}}$\\[3pt]
    \textbf{[P2] Env Field Autoregression:}\\
    $E_{y,x}(t) = \text{Norm}\left(\rho E_{y,x}(t-1) + (1-\rho) Z_{\text{smooth}}\right)$\\[3pt]
    \textbf{[P3] Patch Disturbance:}\\
    $I^{\text{dist}}_{y,x,k} = I^{\text{new}}_{y,x,k} (1 - S_{\text{dist}} E_{y,x}) \quad \text{for } d(y,x; c_y,c_x) < R_{\text{dist}}$\\[3pt]
    \textbf{[P4] Niche Fitness (Mahalanobis):}\\
    $w_i = \text{clip}\left(1 - \frac{(\boldsymbol{q}_i - \boldsymbol{c}_{s_i})^T \boldsymbol{\Sigma}_{s_i}^{-1} (\boldsymbol{q}_i - \boldsymbol{c}_{s_i})}{\sigma_{\text{thresh}}^2}, 0, 1\right)$
};

\node[procbox, right=0.8cm of p1_p4, text width=8.0cm] (p5_p7) {
    \textbf{Processes 5--7: Uptake, Limitation \& Growth}\\
    \hrulefill\\[2pt]
    \textbf{[P5] Michaelis-Menten Nutrient Uptake:}\\
    $U_{i,k} = U^{\text{des}}_{i,k} \cdot \frac{I^{\text{avail}}_{y_i, x_i, k}}{I^{\text{avail}}_{y_i, x_i, k} + K_m}, \quad K_m = 0.1$\\[3pt]
    \textbf{[P6] Liebig Quota \& Space Constraints:}\\
    $g_i^{\text{lim}} = \min_{e} \left(\max\left(0, 1 - \frac{Q_{\min}}{Q_{i,e}}\right)\right), \quad f_i^{\text{space}} = \max\left(0, 1 - \frac{B_{y_i,x_i}}{K_{\text{bio}}}\right)$\\[3pt]
    \textbf{[P7] Realized Growth \& Respiration:}\\
    $G_i^{\text{real}} = \left(U_{i,\text{C}} + \sum_k U_{i,k}\right) g_i^{\text{lim}} f_i^{\text{space}}$\\
    $m_i'' = m_i + G_i^{\text{real}} - m_i r_{\text{resp}} (1 + \beta_r(T_{y_i,x_i}-0.5)) + \xi_i, \;\; \xi_i \sim \mathcal{N}(0, \sigma_{\text{demo}}^2)$
};

\node[procbox, below=0.6cm of p1_p4, text width=7.2cm] (p8) {
    \textbf{Process 8: Multi-Tier Mortality Filters}\\
    \hrulefill\\[2pt]
    Viability: $\alpha_i^{(0)} = \mathbb{I}(m_i'' > 0.01)$\\[2pt]
    Catastrophe: $P_i^{\text{cat}} = d_{\text{cat}} S_{y_i,x_i} \quad \text{if } t \pmod{T_{\text{cat}}} == 0$\\[2pt]
    Weak Disturbance: $P_i^{\text{weak}} = d_{\text{weak}} E_{y_i,x_i} (1 - w_i)^2$\\[2pt]
    Strong Disturbance: $P_i^{\text{strong}} = d_{\text{strong}} S_{y_i,x_i} (\theta_i)^2, \quad \theta_i \in [0,1]$
};

\node[procbox, below=0.6cm of p5_p7, text width=8.0cm] (p9_p10) {
    \textbf{Processes 9--10: Recycling \& Reproduction}\\
    \hrulefill\\[2pt]
    \textbf{[P9] Turnover, Recycling \& Mineralization:}\\
    $L_{i,e} = \left(m_i^{\text{dead}} + m_i^{\text{alive}} \mu_{\text{turn}}\right) Q_{i,e} \longrightarrow O_{y,x,k}^{\text{tot}}$\\
    $\Phi^{\text{min}}_{y,x,k} = O^{\text{tot}}_{y,x,k} m_{\text{min}} \left(1 + \beta_m(T_{y,x}-0.5)\right) \longrightarrow I_{y,x,k}(t+1)$\\[3pt]
    \textbf{[P10] Dispersal \& Toroidal Establishment:}\\
    $y_{\text{off}} = (y_i + \Delta y) \pmod H, \quad x_{\text{off}} = (x_i + \Delta x) \pmod W$\\
    $P^{\text{est}} = \max\left(0, 1 - \frac{B_{y_{\text{off}}, x_{\text{off}}}}{K_{\text{bio}}}\right) \quad \text{with } P^{\text{seed}} = 0.1 w_i$
};
